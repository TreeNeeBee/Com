cmake_minimum_required ( VERSION "3.10.2" )

project ( ComDdsBindingTest )

set ( CMAKE_CXX_STANDARD 17 )
set ( CMAKE_CXX_STANDARD_REQUIRED ON )

# Find dependencies
find_package( nlohmann_json REQUIRED )
find_package( fastdds REQUIRED )
find_package( fastcdr REQUIRED )
find_package( GTest REQUIRED )

# Source directories
set ( COM_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../.. )
set ( CORE_ROOT ${COM_ROOT}/../Core )
set ( LOG_ROOT ${COM_ROOT}/../LogAndTrace )

# Create include symlinks for lap/core and lap/log
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include/lap)

execute_process(
    COMMAND ${CMAKE_COMMAND} -E create_symlink
    ${CORE_ROOT}/source/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include/lap/core
)

execute_process(
    COMMAND ${CMAKE_COMMAND} -E create_symlink
    ${LOG_ROOT}/source/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include/lap/log
)

execute_process(
    COMMAND ${CMAKE_COMMAND} -E create_symlink
    ${CORE_ROOT}/source/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include/core
)

execute_process(
    COMMAND ${CMAKE_COMMAND} -E create_symlink
    ${LOG_ROOT}/source/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include/log
)

include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}/inc
    ${COM_ROOT}/source/binding/common
    ${COM_ROOT}/source/runtime/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include
    ${CORE_ROOT}/source/inc
    ${LOG_ROOT}/source/inc
)

# Link directories
link_directories(
    /usr/local/lib
    ${CORE_ROOT}/build
    ${LOG_ROOT}/build
)

# DDS wire type is now defined in code (CDdsPayload.hpp) — no IDL sources needed.
# Per-service types are generated by the lap-sidl-gen → fastddsgen pipeline and
# registered at runtime via CDdsTypeRegistry.

# Build DDS binding as shared library
add_library( lap_com_binding_dds SHARED
    ${CMAKE_CURRENT_SOURCE_DIR}/src/DdsBinding.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/DdsReaderListener.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/DdsDiscoveryListener.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CDdsCodec.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CDdsServiceManager.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CDdsEventManager.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CDdsMethodManager.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/CDdsDiscoveryServerMonitor.cpp
)

target_link_libraries( lap_com_binding_dds
    lap_core
    lap_log
    fastdds
    fastcdr
    pthread
    rt
    dl
)

# Installation
install( TARGETS lap_com_binding_dds
    LIBRARY DESTINATION lib
)

install( FILES
    ${CMAKE_CURRENT_SOURCE_DIR}/inc/DdsBinding.hpp
    DESTINATION include/lap/com/binding
)

# Discovery test (GTest)
add_executable( dds_discovery_test
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_discovery.cpp
)

target_link_libraries( dds_discovery_test
    lap_com_binding_dds
    lap_core
    lap_log
    fastdds
    fastcdr
    GTest::gtest
    GTest::gtest_main
    pthread
    rt
    dl
)

target_include_directories( dds_discovery_test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/inc
    ${COM_ROOT}/source/binding/common
    ${COM_ROOT}/source/runtime/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include
    ${CORE_ROOT}/source/inc
    ${LOG_ROOT}/source/inc
)

message( STATUS "DDS binding configured successfully" )

# Discovery Server Monitor tests (GTest)
add_executable( dds_ds_monitor_test
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_ds_monitor.cpp
)

target_link_libraries( dds_ds_monitor_test
    lap_com_binding_dds
    lap_core
    lap_log
    fastdds
    fastcdr
    GTest::gtest
    GTest::gtest_main
    pthread
    rt
    dl
)

target_include_directories( dds_ds_monitor_test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/inc
    ${COM_ROOT}/source/binding/common
    ${COM_ROOT}/source/runtime/inc
    ${CMAKE_CURRENT_BINARY_DIR}/include
    ${CORE_ROOT}/source/inc
    ${LOG_ROOT}/source/inc
)
